import UIKit import CoreMotion class HeightMeasurementViewController: UIViewController { // UI Components @IBOutlet weak var unitSegmentControl: UISegmentedControl! @IBOutlet weak var startButton: UIButton! @IBOutlet weak var resultLabel: UILabel! @IBOutlet weak var statusLabel: UILabel! // 센서 및 측정 관련 변수 let motionManager = CMMotionManager() var startY: CGFloat = 0 var endY: CGFloat = 0 var startAttitude: CMAttitude? var endAttitude: CMAttitude? var isMeasuring = false // 캘리브레이션 변수 var calibrationFactor: Double = 1.0 // 픽셀 to cm 변환 계수 var isCalibrated = false override func viewDidLoad() { super.viewDidLoad() setupMotionManager() setupUI() showCalibrationAlert() } // MARK: - Motion Manager Setup func setupMotionManager() { guard motionManager.isDeviceMotionAvailable else { statusLabel.text = "기기가 모션 센서를 지원하지 않습니다" return } motionManager.deviceMotionUpdateInterval = 0.01 // 100Hz motionManager.startDeviceMotionUpdates() } // MARK: - UI Setup func setupUI() { startButton.setTitle("측정 시작", for: .normal) startButton.addTarget(self, action: #selector(toggleMeasurement), for: .touchUpInside) unitSegmentControl.setSegmentAction(for: .valueChanged) { [weak self] in self?.updateResultDisplay() } resultLabel.text = "준비됨" statusLabel.text = "먼저 캘리브레이션을 진행하세요" } // MARK: - Calibration func showCalibrationAlert() { let alert = UIAlertController(title: "캘리브레이션", message: "핸드폰을 정확히 수직으로 놓고 '10cm 올리기'를 선택한 후, 핸드폰을 정확히 10cm 올렸을 때 '완료'를 누르세요.", preferredStyle: .alert) alert.addAction(UIAlertAction(title: "취소", style: .cancel)) alert.addAction(UIAlertAction(title: "시작", style: .default) { [weak self] _ in self?.startCalibration() }) present(alert, animated: true) } func startCalibration() { isMeasuring = true startY = 0 startAttitude = motionManager.deviceMotion?.attitude.copy() as? CMAttitude statusLabel.text = "10cm 올리고 있습니다... (기다리세요)" startButton.isEnabled = false } @objc func completeCalibration() { guard let currentMotion = motionManager.deviceMotion else { return } endY = 100 // 10cm = 100mm (실제로 핸드폰을 10cm 올렸다고 가정) endAttitude = currentMotion.attitude.copy() as? CMAttitude calculateCalibrationFactor() } func calculateCalibrationFactor() { guard let startAtt = startAttitude, let endAtt = endAttitude else { return } let yawDifference = endAtt.yaw - startAtt.yaw let pitchDifference = endAtt.pitch - startAtt.pitch let rollDifference = endAtt.roll - startAtt.roll // Y축 변화량에 영향을 주는 각도 변화 (pitch와 roll) let angleCorrection = sqrt(pitchDifference * pitchDifference + rollDifference * rollDifference) // 캘리브레이션: 10cm 올렸을 때의 Y좌표 변화량으로부터 계수 계산 calibrationFactor = 100.0 / max(abs(endY - startY), 1.0) // 10cm를 픽셀 변화로 정규화 // 각도 보정 적용 calibrationFactor *= (1.0 + angleCorrection) isCalibrated = true isMeasuring = false startButton.isEnabled = true statusLabel.text = "캘리브레이션 완료! 이제 측정을 시작할 수 있습니다." let alert = UIAlertController(title: "캘리브레이션 완료", message: "캘리브레이션이 완료되었습니다. 이제 높이를 측정할 수 있습니다.", preferredStyle: .alert) alert.addAction(UIAlertAction(title: "확인", style: .default)) present(alert, animated: true) } // MARK: - Height Measurement @objc func toggleMeasurement() { guard isCalibrated else { statusLabel.text = "먼저 캘리브레이션을 완료하세요" return } if !isMeasuring { startMeasurement() } else { endMeasurement() } } func startMeasurement() { isMeasuring = true startY = 0 // 터치 위치의 Y좌표 startAttitude = motionManager.deviceMotion?.attitude.copy() as? CMAttitude startButton.setTitle("측정 완료", for: .normal) statusLabel.text = "높이를 측정 중입니다... (측정 완료 버튼을 누르세요)" resultLabel.text = "측정 중..." } func endMeasurement() { guard let currentMotion = motionManager.deviceMotion else { return } isMeasuring = false endY = 0 // 터치 위치의 Y좌표 (올린 후) endAttitude = currentMotion.attitude.copy() as? CMAttitude startButton.setTitle("측정 시작", for: .normal) calculateHeight() } func calculateHeight() { guard let startAtt = startAttitude, let endAtt = endAttitude else { statusLabel.text = "센서 데이터를 읽을 수 없습니다" return } // Y좌표 변화량 let yDifference = abs(endY - startY) // 각도 변화 계산 let yawDifference = endAtt.yaw - startAtt.yaw let pitchDifference = endAtt.pitch - startAtt.pitch let rollDifference = endAtt.roll - startAtt.roll // 각도 변화 크기 let angleChange = sqrt(pitchDifference * pitchDifference + rollDifference * rollDifference) // 높이 계산 (mm 단위) var heightMM = yDifference * calibrationFactor // 각도 보정 적용 if angleChange > 0 { heightMM /= (1.0 + angleChange * 0.5) // 각도 변화에 따른 보정 } // 오차 범위 내에서 반올림 heightMM = round(heightMM) // ±1mm 범위 updateResultDisplay(heightMM: heightMM) } func updateResultDisplay(heightMM: Double? = nil) { if let heightMM = heightMM { let unit = unitSegmentControl.selectedSegmentIndex == 0 ? "cm" : "mm" let displayValue = unit == "cm" ? heightMM / 10.0 : heightMM resultLabel.text = String(format: "높이: %.1f %@", displayValue, unit) statusLabel.text = "측정 완료" } } deinit { motionManager.stopDeviceMotionUpdates() } }
